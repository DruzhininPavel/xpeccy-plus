# chapter 5: CIPSENDBUF, CIPBUFSTATUS, CIPCHECKSEQ, CIPCHECKQUEUE, CIPBUFRESET (MUX 0 and 1)
mask .*\\x[0-9A-F][0-9A-F].* <romlog>
wait WIFI GOT IP\r\n 5000
send ATE0
wait \r\nOK\r\n
server echo
send AT+CIPSENDBUF=4
wait \r\nERROR\r\n
send AT+CIPSENDBUF=?
wait \r\nERROR\r\n
send AT+CIPBUFSTATUS
wait \r\nOK\r\n
send AT+CIPBUFRESET
wait \r\nERROR\r\n
send AT+CIPSTART="TCP","127.0.0.1",<port>
wait \r\nOK\r\n
send AT+CIPBUFSTATUS
wait \r\nOK\r\n
send AT+CIPBUFSTATUS=0
wait \r\nERROR\r\n
send AT+CIPSENDBUF=3000
wait \r\nERROR\r\n
send AT+CIPSENDBUF=4
wait >\x20
raw abcd
wait SEND OK\r\n
quiet 300
send AT+CIPCHECKSEQ=1
wait \r\nOK\r\n
send AT+CIPCHECKQUEUE=1
wait \r\nOK\r\n
send AT+CIPCHECKSEQ=2
wait \r\nERROR\r\n
send AT+CIPCHECKSEQ=0
wait \r\nERROR\r\n
send AT+CIPBUFSTATUS
wait \r\nOK\r\n
send AT+CIPSENDBUF=2
wait >\x20
raw xy
wait SEND OK\r\n
quiet 300
send AT+CIPBUFSTATUS
wait \r\nOK\r\n
send AT+CIPBUFRESET=0
wait \r\nERROR\r\n
send AT+CIPBUFRESET
wait \r\nOK\r\n
send AT+CIPBUFSTATUS
wait \r\nOK\r\n
send AT+CIPCLOSE
wait \r\nOK\r\n
# MUX 1: the id in front
send AT+CIPMUX=1
wait \r\nOK\r\n
send AT+CIPSTART=3,"TCP","127.0.0.1",<port>
wait \r\nOK\r\n
send AT+CIPBUFSTATUS
wait \r\nERROR\r\n
send AT+CIPSENDBUF=3,2
wait >\x20
raw ok
wait SEND OK\r\n
quiet 300
send AT+CIPBUFSTATUS=3
wait \r\nOK\r\n
send AT+CIPCHECKSEQ=3,1
wait \r\nOK\r\n
send AT+CIPCHECKSEQ=1
wait \r\nERROR\r\n
send AT+CIPBUFRESET=3
wait \r\nOK\r\n
send AT+CIPCLOSE=3
wait \r\nOK\r\n
