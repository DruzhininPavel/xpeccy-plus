# chapter 5: UDP - CONNECT and SEND OK at once, CIPDINFO=1 naming each datagram's sender, a
# CIPSEND target (an address, a name, one not found) that stays the peer, modes 0/1/2, MUX 1
mask .*\\x[0-9A-F][0-9A-F].* <romlog>
wait WIFI GOT IP\r\n 5000
send ATE0
wait \r\nOK\r\n
send AT+CIPDINFO=1
wait \r\nOK\r\n
# mode 0: the peer stays what CIPSTART said
send AT+CIPSTART="UDP","127.0.0.1",<uport>,5000,0
wait \r\nOK\r\n
send AT+CIPSEND=5
wait >\x20
raw hello
wait SEND OK\r\n
pause 100
udpshow
udp 5000 back
wait back
send AT+CIPSTATUS
wait \r\nOK\r\n
send AT+CIPSEND=2,"127.0.0.1",9
wait >\x20
raw xx
wait SEND OK\r\n
send AT+CIPSTATUS
wait \r\nOK\r\n
send AT+CIPSEND=2
wait >\x20
raw yy
wait SEND OK\r\n
pause 100
udpshow
send AT+CIPSEND=2,"localhost",<uport>
wait >\x20
raw zz
wait SEND OK\r\n
pause 100
udpshow
send AT+CIPSEND=2,"nonexistent.invalid",5
wait \r\nERROR\r\n 15000
send AT+CIPCLOSE
wait \r\nOK\r\n
# mode 1: the peer becomes the first sender, then stays
send AT+CIPSTART="UDP","127.0.0.1",9,5001,1
wait \r\nOK\r\n
udp 5001 a1
wait a1
send AT+CIPSTATUS
wait \r\nOK\r\n
send AT+CIPSEND=2
wait >\x20
raw m1
wait SEND OK\r\n
pause 100
udpshow
send AT+CIPMODE=1
wait \r\nOK\r\n
send AT+CIPMODE=0
wait \r\nOK\r\n
send AT+CIPCLOSE
wait \r\nOK\r\n
# mode 2: every sender; no passthrough with it
send AT+CIPSTART="UDP","127.0.0.1",9,5002,2
wait \r\nOK\r\n
send AT+CIPMODE=1
wait \r\nERROR\r\n
udp 5002 b2
wait b2
send AT+CIPSEND=2
wait >\x20
raw m2
wait SEND OK\r\n
pause 100
udpshow
send AT+CIPMODE=1
wait \r\nERROR\r\n
send AT+CIPDINFO=0
wait \r\nOK\r\n
udp 5002 c3
wait c3
send AT+CIPCLOSE
wait \r\nOK\r\n
# MUX 1
send AT+CIPMUX=1
wait \r\nOK\r\n
send AT+CIPSTART=4,"UDP","127.0.0.1",<uport>,5003,0
wait \r\nOK\r\n
udp 5003 dd
wait dd
send AT+CIPSEND=4,2
wait >\x20
raw ee
wait SEND OK\r\n
pause 100
udpshow
send AT+CIPCLOSE=4
wait \r\nOK\r\n
