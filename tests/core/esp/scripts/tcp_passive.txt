# chapter 5: passive receive - CIPRECVMODE, the +IPD notice, CIPRECVLEN, CIPRECVDATA, data kept after a close
mask .*\\x[0-9A-F][0-9A-F].* <romlog>
wait WIFI GOT IP\r\n 5000
send ATE0
wait \r\nOK\r\n
server echo
send AT+CIPRECVMODE?
wait \r\nOK\r\n
send AT+CIPRECVMODE=2
wait \r\nERROR\r\n
send AT+CIPRECVMODE=1
wait \r\nOK\r\n
send AT+CIPRECVMODE?
wait \r\nOK\r\n
send AT+CIPRECVLEN?
wait \r\nOK\r\n
send AT+CIPRECVLEN
wait \r\nERROR\r\n
send AT+CIPRECVDATA=5
wait \r\nERROR\r\n
send AT+CIPSTART="TCP","127.0.0.1",<port>
wait \r\nOK\r\n
send AT+CIPSEND=5
wait >\x20
raw hello
wait +IPD,5\r\n
send AT+CIPRECVLEN?
wait \r\nOK\r\n
send AT+CIPRECVDATA=0
wait \r\nERROR\r\n
send AT+CIPRECVDATA=2049
wait \r\nERROR\r\n
send AT+CIPRECVDATA=3
wait \r\nOK\r\n
send AT+CIPRECVDATA=10
wait \r\nOK\r\n
send AT+CIPRECVDATA=10
wait \r\nOK\r\n
send AT+CIPSEND=4
wait >\x20
raw data
wait +IPD,4\r\n
send AT+CIPCLOSE
wait \r\nOK\r\n
send AT+CIPRECVLEN?
wait \r\nOK\r\n
send AT+CIPRECVDATA=10
wait \r\nOK\r\n
send AT+CIPRECVDATA=10
wait \r\nERROR\r\n
send AT+CIPRECVMODE=0
wait \r\nOK\r\n
