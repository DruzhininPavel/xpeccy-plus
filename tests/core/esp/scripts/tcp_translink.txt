# chapter 5: SAVETRANSLINK - its checks, then a reset: the module dials the saved link and is in
# passthrough at once (its Wi-Fi lines kept quiet), till +++. Run with FLASH set.
mask .*\\x[0-9A-F][0-9A-F].* <romlog>
mask ,\d{5},0\\r\\n$ ,<lport>,0\r\n
wait WIFI GOT IP\r\n 5000
send ATE0
wait \r\nOK\r\n
send AT+SAVETRANSLINK?
wait \r\nERROR\r\n
send AT+SAVETRANSLINK=2,"127.0.0.1",<port>
wait \r\nERROR\r\n
send AT+SAVETRANSLINK=1,"127.0.0.1",0
wait \r\nERROR\r\n
send AT+SAVETRANSLINK=1,"0.0.0.0",<port>
wait \r\nERROR\r\n
send AT+SAVETRANSLINK=1,127.0.0.1,<port>
wait \r\nERROR\r\n
send AT+SAVETRANSLINK=1,"127.0.0.1",<port>,"FOO"
wait \r\nERROR\r\n
send AT+SAVETRANSLINK=1,"127.0.0.1",<port>,"UDP",70000
wait \r\nERROR\r\n
send AT+SAVETRANSLINK=1,"127.0.0.1",<port>,"TCP",8000
wait \r\nERROR\r\n
send AT+SAVETRANSLINK=1,"127.0.0.1",<port>,"TCP",60
wait \r\nOK\r\n
server echo
send AT+RST
wait ready\r\n 3000
pause 4000
raw ping
wait ping 3000
pause 500
raw +++
pause 1500
send AT+CIPMODE?
wait \r\nOK\r\n
send AT+CIPSTATUS
wait \r\nOK\r\n
send AT+SAVETRANSLINK=0
wait \r\nOK\r\n
send AT+CIPMODE=0
wait \r\nOK\r\n
send AT+CIPCLOSE
wait \r\nOK\r\n
