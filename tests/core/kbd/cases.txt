# Keyboard bench cases: a host key that is a Spectrum combination (. = SS+M, Delete =
# CS+0) has its shift down before the key and up after it; ours go whole.
#
#   bin  <file in bin/>  <source in asm/>         how mkbins.py assembles it
#   run  <case>  <file in bin/>  <presses>  [ENV=value ...]

bin  poll.bin  poll.asm

run  48k-host   poll.bin  20  MACHINE=ZX48 KEYS=host
run  48k-ours   poll.bin  20  MACHINE=ZX48 KEYS=ours
run  pent-host  poll.bin  20  MACHINE=Pentagon KEYS=host
run  next-host  poll.bin  20  MACHINE=ZXNext KEYS=host
